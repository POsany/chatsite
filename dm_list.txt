<?php
session_start();
require_once 'db_connect.php';

if (!isset($_SESSION['user_id'])) {
    header('Location: login.php');
    exit;
}

$my_id = $_SESSION['user_id'];

// 最新のメッセージを送信者ごとに取得
$sql = "SELECT m.*, u.nickname, u.profile_image 
        FROM direct_messages m
        JOIN users u ON (m.sender_id = u.id OR m.receiver_id = u.id) AND u.id != :my_id
        WHERE m.id IN (
            SELECT MAX(id) FROM direct_messages 
            WHERE sender_id = :my_id OR receiver_id = :my_id 
            GROUP BY IF(sender_id = :my_id, receiver_id, sender_id)
        )
        ORDER BY m.created_at DESC";

$stmt = $pdo->prepare($sql);
$stmt->execute([':my_id' => $my_id]);
$chats = $stmt->fetchAll();
?>
